<!-- The form itself -->
<form id="userForm">
  <input type="text" name="name" placeholder="Full Name" required>
  <input type="email" name="email" placeholder="Email" required>
  <input type="tel" name="phone" placeholder="Phone Number">
  <input type="text" name="country" placeholder="Country">
  <textarea name="message" placeholder="Initial message (optional)"></textarea>
  <button type="submit">Register</button>
</form>

<!-- Status message area -->
<div id="formStatus"></div>

<script>
// 🔴 REPLACE THIS URL WITH YOUR ACTUAL WEB APP URL 🔴
const API_URL = 'https://script.google.com/macros/s/YOUR_ID_HERE/exec';

// Get the form element
const form = document.getElementById('userForm');

// Add submit event listener
form.addEventListener('submit', async (event) => {
  event.preventDefault(); // Stop page from refreshing
  
  // Show loading status
  const statusDiv = document.getElementById('formStatus');
  statusDiv.innerHTML = 'Saving your data...';
  statusDiv.style.color = 'blue';
  
  // Collect all form data
  const formData = new FormData(form);
  const userData = {};
  formData.forEach((value, key) => {
    userData[key] = value;
  });
  
  // Add timestamp automatically
  userData.timestamp = new Date().toISOString();
  userData.chatHistory = ''; // Will be filled later
  userData.lastActive = new Date().toISOString();
  
  try {
    // Send data to Google Sheet
    const response = await fetch(`${API_URL}?sheet=Users`, {
      method: 'POST',
      body: JSON.stringify(userData)
    });
    
    const result = await response.json();
    
    if (result.success) {
      statusDiv.innerHTML = '✅ Registration successful! Your data has been saved.';
      statusDiv.style.color = 'green';
      form.reset(); // Clear the form
    } else {
      statusDiv.innerHTML = '❌ Error: ' + (result.error || 'Unknown error');
      statusDiv.style.color = 'red';
    }
  } catch (error) {
    statusDiv.innerHTML = '❌ Connection error: ' + error.message;
    statusDiv.style.color = 'red';
  }
});
</script>
